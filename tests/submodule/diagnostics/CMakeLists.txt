cmake_minimum_required(VERSION 3.25)
project(diagnostics_config_consumer LANGUAGES C)

if(NOT DEFINED ENGINE_ROOT)
    message(FATAL_ERROR "ENGINE_ROOT must point to the engine source directory")
endif()
add_subdirectory("${ENGINE_ROOT}" "${CMAKE_BINARY_DIR}/engine" EXCLUDE_FROM_ALL)

set(NT426_EXPECT_LOG ${NT_LOG_MIN_LEVEL})
foreach(_feature UI_TIMING GFX_GPU_TIMING INTROSPECT INTROSPECT_WRITE METRICS LOG_RING)
    if(NT_${_feature}_ENABLED)
        set(NT426_EXPECT_${_feature} 1)
    else()
        set(NT426_EXPECT_${_feature} 0)
    endif()
endforeach()
configure_file(expected.h.in expected.h @ONLY)

function(nt426_consumer name check)
    add_library(${name} OBJECT "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/consumer.c")
    target_compile_features(${name} PRIVATE c_std_17)
    target_compile_definitions(${name} PRIVATE NT426_CHECK_${check}=1)
    target_include_directories(${name} PRIVATE "${CMAKE_BINARY_DIR}")
    target_link_libraries(${name} PRIVATE ${ARGN})
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${name} PRIVATE -Werror -Wundef)
    endif()
endfunction()

nt426_consumer(parent_log LOG nt_log_interface)
nt426_consumer(parent_core CORE nt_core)
nt426_consumer(parent_ui UI nt_ui_interface nt_shared)
nt426_consumer(parent_gfx GFX nt_gfx_interface nt_shared)
add_subdirectory(sibling)

foreach(_impl real stub)
    add_executable(log_${_impl} log_link.c)
    target_include_directories(log_${_impl} PRIVATE "${CMAKE_BINARY_DIR}")
    nt_set_static_crt(log_${_impl})
    nt_set_sanitizer_flags(log_${_impl})
    if(_impl STREQUAL "real")
        target_compile_definitions(log_${_impl} PRIVATE NT426_LOG_STUB=0)
        target_link_libraries(log_${_impl} PRIVATE nt_log)
    else()
        target_compile_definitions(log_${_impl} PRIVATE NT426_LOG_STUB=1)
        target_link_libraries(log_${_impl} PRIVATE nt_log_stub)
    endif()
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/log_${_impl}.path" CONTENT "$<TARGET_FILE:log_${_impl}>")
endforeach()

add_executable(log_missing EXCLUDE_FROM_ALL log_missing.c)
target_link_libraries(log_missing PRIVATE nt_log_interface)

add_custom_target(diagnostics_config ALL DEPENDS
    parent_log parent_core parent_ui parent_gfx
    sibling_core sibling_component sibling_metrics sibling_ring sibling_ui_stub log_real log_stub)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/log_archive.path" CONTENT "$<TARGET_FILE:nt_log>")
